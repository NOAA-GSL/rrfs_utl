cmake_minimum_required(VERSION 3.18)

project(
  raymond
  LANGUAGES C Fortran
)

include(GNUInstallDirs)

# Path to the Fortran source
set(FCODE "${CMAKE_CURRENT_SOURCE_DIR}/raymond.f")

# Python executable
if(DEFINED ENV{BLENDINGPYTHON})
  set(BPY "$ENV{BLENDINGPYTHON}")
else()
  find_program(BPY python3)
endif()

if(NOT BPY OR NOT EXISTS "${BPY}")
  message(FATAL_ERROR "Unable to locate a valid Python executable for building raymond (BPY='${BPY}')")
endif()

# Force CMake to use the same Python selected for the blending build.
set(
  Python_EXECUTABLE
  "${BPY}"
  CACHE FILEPATH
  "Python executable used to build raymond"
  FORCE
)

# Find Python, the Python module development headers, and NumPy.
find_package(
  Python 3.12 REQUIRED
  COMPONENTS Interpreter Development.Module NumPy
)

# Locate the NumPy F2PY support sources, including fortranobject.c.
execute_process(
  COMMAND
    "${Python_EXECUTABLE}"
    -c
    "import numpy.f2py; print(numpy.f2py.get_include())"
  RESULT_VARIABLE F2PY_INCLUDE_RESULT
  OUTPUT_VARIABLE F2PY_INCLUDE_DIR
  ERROR_VARIABLE F2PY_INCLUDE_ERROR
  OUTPUT_STRIP_TRAILING_WHITESPACE
)

if(NOT F2PY_INCLUDE_RESULT EQUAL 0)
  message(
    FATAL_ERROR
    "Unable to locate the NumPy F2PY include directory:\n"
    "${F2PY_INCLUDE_ERROR}"
  )
endif()

# Print the Python and NumPy environment selected by CMake.
execute_process(
  COMMAND
    "${Python_EXECUTABLE}"
    -c
    "import sys, numpy; print(sys.executable); print(sys.version); print(numpy.__version__); print(numpy.__file__)"
  RESULT_VARIABLE PYTHON_INFO_RESULT
  OUTPUT_VARIABLE PYTHON_INFO
  ERROR_VARIABLE PYTHON_INFO_ERROR
  OUTPUT_STRIP_TRAILING_WHITESPACE
)

if(NOT PYTHON_INFO_RESULT EQUAL 0)
  message(
    FATAL_ERROR
    "Unable to query the blending Python environment:\n"
    "${PYTHON_INFO_ERROR}"
  )
endif()

message(STATUS "BLENDINGPYTHON: ${Python_EXECUTABLE}")
message(STATUS "Blending Python information:\n${PYTHON_INFO}")
message(STATUS "Python include directories: ${Python_INCLUDE_DIRS}")
message(STATUS "NumPy include directories: ${Python_NumPy_INCLUDE_DIRS}")
message(STATUS "F2PY include directory: ${F2PY_INCLUDE_DIR}")

# Files generated by F2PY.
set(F2PY_MODULE_C "${CMAKE_CURRENT_BINARY_DIR}/raymondmodule.c")
set(F2PY_WRAPPER_F "${CMAKE_CURRENT_BINARY_DIR}/raymond-f2pywrappers.f")

# Use F2PY only to generate wrapper sources.
#
# Do not pass -c here. The -c option invokes the Meson compilation backend
# with Python 3.12. CMake will compile the generated sources directly.
add_custom_command(
  OUTPUT
    "${F2PY_MODULE_C}"
    "${F2PY_WRAPPER_F}"
  COMMAND
    "${Python_EXECUTABLE}"
    -m numpy.f2py
    "${FCODE}"
    -m raymond
    --lower
  WORKING_DIRECTORY "${CMAKE_CURRENT_BINARY_DIR}"
  DEPENDS "${FCODE}"
  COMMENT "Generating raymond F2PY wrapper sources"
  VERBATIM
)

# Group the generated files into a target so the extension library depends
# on wrapper generation.
add_custom_target(
  generate_raymond_f2py
  DEPENDS
    "${F2PY_MODULE_C}"
    "${F2PY_WRAPPER_F}"
)

# Build the Python extension with CMake rather than the F2PY Meson backend.
Python_add_library(
  raymond
  MODULE
  "${F2PY_MODULE_C}"
  "${F2PY_WRAPPER_F}"
  "${F2PY_INCLUDE_DIR}/fortranobject.c"
  "${FCODE}"
)

add_dependencies(raymond generate_raymond_f2py)

# Python::NumPy supplies the NumPy include directories and Python module
# integration required by the generated F2PY C wrapper.
target_link_libraries(
  raymond
  PRIVATE
  Python::NumPy
)

target_include_directories(
  raymond
  PRIVATE
  "${F2PY_INCLUDE_DIR}"
)

# Preserve the existing NumPy C API selection.
target_compile_definitions(
  raymond
  PRIVATE
  NPY_NO_DEPRECATED_API=NPY_1_7_API_VERSION
)

# Preserve the Intel diagnostic suppression flags from the original build.
#
# The original NumPy 1.20.1 F2PY build used the "intelem" compiler
# configuration, which added:
#
#   -fp-model strict -O1 -qopenmp
#
# Reproduce those flags so the Python 3.12 build uses the same Fortran
# floating-point and optimization behavior as the Python 3.8 build.
target_compile_options(
  raymond
  PRIVATE
  $<$<COMPILE_LANGUAGE:C>:-diag-disable=2650>
  $<$<COMPILE_LANGUAGE:C>:-diag-disable=1478>
  $<$<COMPILE_LANGUAGE:Fortran>:-diag-disable=2650>
  $<$<COMPILE_LANGUAGE:Fortran>:-diag-disable=1478>
  $<$<COMPILE_LANGUAGE:Fortran>:-fp-model>
  $<$<COMPILE_LANGUAGE:Fortran>:strict>
  $<$<COMPILE_LANGUAGE:Fortran>:-O1>
  $<$<COMPILE_LANGUAGE:Fortran>:-qopenmp>
)

# The original F2PY build also linked with Intel OpenMP support.
target_link_options(
  raymond
  PRIVATE
  -qopenmp
)

# Produce raymond.so so the existing "import raymond" behavior and installed
# filename remain unchanged.
set_target_properties(
  raymond
  PROPERTIES
  PREFIX ""
  OUTPUT_NAME "raymond"
  SUFFIX ".so"
)

# Install the resulting extension.
install(
  TARGETS raymond
  LIBRARY DESTINATION "${CMAKE_INSTALL_LIBDIR}"
)
